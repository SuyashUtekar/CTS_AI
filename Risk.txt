# ── Risk Score Calculator ─────────────────────────────────────────────────────

def calculate_risk_score(sig: ModelScore, tamper: ModelScore,
anomaly: ModelScore, behaviour: ModelScore,
idp_flags: List[str]) -> int:
“””
Weighted ensemble to produce final risk score 0-100.
Weights: Signature 30%, Tamper 30%, Anomaly 20%, Behaviour 10%, IDP 10%
“””
# Normalize each score to risk contribution (0-100)
sig_risk      = (1 - sig.score)     * 100 * 0.30   # low match = high risk
tamper_risk   = tamper.score        * 100 * 0.30   # high tamper = high risk
anomaly_risk  = anomaly.score       * 100 * 0.20
behaviour_risk= behaviour.score     * 100 * 0.10

```
# IDP flags add fixed risk
idp_risk = min(len(idp_flags) * 15, 30) * 0.10

total = sig_risk + tamper_risk + anomaly_risk + behaviour_risk + idp_risk
return min(100, max(0, int(total)))
```

def get_risk_level(score: int) -> RiskLevel:
if score <= 30:
return RiskLevel.LOW
elif score <= 70:
return RiskLevel.MEDIUM
return RiskLevel.HIGH

def get_risk_factors(sig: ModelScore, tamper: ModelScore,
anomaly: ModelScore, behaviour: ModelScore) -> tuple:
“”“Return top risk and safe factors for explainability”””
risk_factors = []
safe_factors = []

```
if sig.verdict != "GENUINE":
    risk_factors.append("Signature mismatch detected")
else:
    safe_factors.append("Signature verified as genuine")

if tamper.verdict != "NO_TAMPERING":
    risk_factors.append("Possible image tampering detected")
else:
    safe_factors.append("No tampering detected")

if anomaly.verdict != "NORMAL":
    risk_factors.append(f"Amount anomaly: {anomaly.verdict}")
else:
    safe_factors.append("Amount within normal range")

if behaviour.verdict != "NORMAL":
    risk_factors.append("Unusual account behaviour pattern")
else:
    safe_factors.append("Normal account behaviour")

return risk_factors, safe_factors
```

# ── Main Risk Pipeline ────────────────────────────────────────────────────────

# Initialize models once

sig_verifier   = SignatureVerifier()
tamper_detector = TamperDetector()
anomaly_model  = AmountAnomalyDetector()
behaviour_model = BehaviourModel()

def run_risk_scoring(cheque_id: str, signature_path: str,
image_path: str, fields: OCRFields,
idp_flags: list) -> RiskResult:
“”“Run full risk scoring pipeline”””
log.info(f”[Stage 5] Risk scoring for: {cheque_id}”)

```
amount = float(fields.amount_numeric or 0)

# Run all models
sig_result       = sig_verifier.run(signature_path, fields.account_number or "")
tamper_result    = tamper_detector.run(image_path)
anomaly_result   = anomaly_model.run(amount, fields.account_number or "")
behaviour_result = behaviour_model.run(fields.account_number or "", amount)

# Calculate composite risk score
risk_score = calculate_risk_score(sig_result, tamper_result,
                                   anomaly_result, behaviour_result,
                                   idp_flags)
risk_level = get_risk_level(risk_score)

# Get explainability factors
risk_factors, safe_factors = get_risk_factors(
    sig_result, tamper_result, anomaly_result, behaviour_result
)

log.info(f"Risk score for {cheque_id}: {risk_score} ({risk_level})")

return RiskResult(
    cheque_id          = cheque_id,
    risk_score         = risk_score,
    risk_level         = risk_level,
    signature_check    = sig_result,
    tamper_check       = tamper_result,
    amount_anomaly     = anomaly_result,
    account_behaviour  = behaviour_result,
    top_risk_factors   = risk_factors,
    top_safe_factors   = safe_factors
)
```

# ── FastAPI Endpoint ───────────────────────────────────────────────────────────

class RiskRequest(BaseModel):
cheque_id: str
signature_path: str
image_path: str
fields: OCRFields
idp_flags: list = []

@router.post(”/score”, response_model=RiskResult)
async def score_risk(req: RiskRequest):
“”“Run risk scoring on cheque”””
try:
result = run_risk_scoring(
req.cheque_id, req.signature_path,
req.image_path, req.fields, req.idp_flags
)
return result
except Exception as e:
log.error(f”Risk scoring failed for {req.cheque_id}: {e}”)
raise HTTPException(500, str(e))
