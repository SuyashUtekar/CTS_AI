@app.post(”/pipeline/run”, summary=“Run full end-to-end cheque clearing pipeline”)
async def run_full_pipeline(
branch_ifsc:  str        = Form(…),
scanner_id:   str        = Form(…),
front_image:  UploadFile = File(…)
):
“””
Submit a cheque image and run the complete 6-stage pipeline:
Ingest → Preprocess → OCR → IDP → Risk → Agent Decision
“””
results = {}
cheque_id = generate_cheque_id()
log.info(f”=== PIPELINE START: {cheque_id} ===”)

```
try:
    # ── Stage 1: Ingest ───────────────────────────────────────────────────
    log.info(f"[1/6] Ingesting cheque...")
    front_path = save_image_locally(cheque_id, front_image, "front")
    results["stage1"] = {
        "cheque_id":        cheque_id,
        "front_image_path": front_path,
        "status":           "INGESTED"
    }

    # ── Stage 2: Preprocess ───────────────────────────────────────────────
    log.info(f"[2/6] Preprocessing image...")
    preprocess_result = preprocess_cheque(cheque_id, front_path)
    results["stage2"] = preprocess_result.model_dump()

    # Build zones dict
    zones = {k: v for k, v in preprocess_result.zones.model_dump().items() if v}

    # ── Stage 3: OCR ──────────────────────────────────────────────────────
    log.info(f"[3/6] Running OCR extraction...")
    ocr_result = run_ocr(cheque_id, zones)
    results["stage3"] = ocr_result.model_dump()

    # ── Stage 4: IDP Validation ───────────────────────────────────────────
    log.info(f"[4/6] Running IDP validation...")
    idp_result = run_idp_validation(cheque_id, ocr_result.fields)
    results["stage4"] = idp_result.model_dump()

    # ── Stage 5: Risk Scoring ─────────────────────────────────────────────
    log.info(f"[5/6] Calculating risk score...")
    sig_path = zones.get("signature_crop", "")
    risk_result = run_risk_scoring(
        cheque_id, sig_path, front_path,
        ocr_result.fields, idp_result.flags
    )
    results["stage5"] = risk_result.model_dump()

    # ── Stage 6: Agent Decision ───────────────────────────────────────────
    log.info(f"[6/6] Running Claude agent decision...")
    if settings.anthropic_api_key:
        decision = run_claude_agent(
            cheque_id, ocr_result.fields, idp_result, risk_result
        )
    else:
        # Fallback rule-based decision if no API key
        from utils.models import Decision, AgentDecision
        score = risk_result.risk_score
        if score <= settings.auto_approve_threshold:
            dec, route = Decision.AUTO_APPROVE, "NGCB_EXPORT"
        elif score <= settings.manual_review_threshold:
            dec, route = Decision.SECONDARY_CHECK, "SECONDARY_CHECK"
        else:
            dec, route = Decision.MANUAL_REVIEW, "MANUAL_REVIEW"

        decision = AgentDecision(
            cheque_id  = cheque_id,
            decision   = dec,
            confidence = 0.80,
            reasoning  = f"Rule-based: risk_score={score}",
            routed_to  = route,
            timestamp  = datetime.now()
        )

    results["stage6"] = decision.model_dump()

    log.info(f"=== PIPELINE COMPLETE: {cheque_id} → {decision.decision} ===")

    return {
        "cheque_id":  cheque_id,
        "final_decision": decision.decision,
        "risk_score": risk_result.risk_score,
        "routed_to":  decision.routed_to,
        "reasoning":  decision.reasoning,
        "all_stages": results
    }

except Exception as e:
    log.error(f"Pipeline failed for {cheque_id}: {e}")
    raise HTTPException(500, f"Pipeline error: {str(e)}")
```
