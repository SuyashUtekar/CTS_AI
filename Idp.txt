# ── Amount in Words → Number Converter ───────────────────────────────────────

ONES  = {“zero”:0,“one”:1,“two”:2,“three”:3,“four”:4,“five”:5,“six”:6,“seven”:7,
“eight”:8,“nine”:9,“ten”:10,“eleven”:11,“twelve”:12,“thirteen”:13,
“fourteen”:14,“fifteen”:15,“sixteen”:16,“seventeen”:17,“eighteen”:18,
“nineteen”:19,“twenty”:20,“thirty”:30,“forty”:40,“fifty”:50,
“sixty”:60,“seventy”:70,“eighty”:80,“ninety”:90}

MULTIPLIERS = {“hundred”:100,“thousand”:1000,“lakh”:100000,“lac”:100000,
“crore”:10000000,“million”:1000000}

def words_to_number(text: str) -> int:
“”“Convert amount-in-words to integer. e.g. ‘Fifty Thousand’ → 50000”””
if not text:
return -1

```
text    = text.lower()
text    = re.sub(r"[^a-z\s]", " ", text)
text    = re.sub(r"\b(rupees?|only|and|rs)\b", "", text)
words   = text.split()

current = 0
result  = 0

for word in words:
    if word in ONES:
        current += ONES[word]
    elif word in MULTIPLIERS:
        mult = MULTIPLIERS[word]
        if mult >= 1000:
            result  += (current if current else 1) * mult
            current  = 0
        else:
            current *= mult

result += current
return result if result > 0 else -1
```

def check_amount_match(fields: OCRFields) -> ValidationCheck:
“”“Verify amount in words matches amount in numbers”””

                               detail=f"Matched: ₹{numeric_val:,}")

# ── Main IDP Validation Pipeline ──────────────────────────────────────────────

def run_idp_validation(cheque_id: str, fields: OCRFields) -> IDPResult:
“”“Run all validation checks and compile result”””
log.info(f”[Stage 4] Running IDP validation for: {cheque_id}”)

```
amount_check  = check_amount_match(fields)
date_check    = check_date(fields)
ifsc_check    = check_ifsc(fields)
micr_check    = check_micr_account_match(fields)

# Collect flags
flags = []
for check_name, check in [
    ("AMOUNT", amount_check),
    ("DATE",   date_check),
    ("IFSC",   ifsc_check),
    ("MICR",   micr_check)
]:
    if check.status == "FAIL":
        flags.append(f"{check_name}_FAIL: {check.detail}")
    elif check.status == "WARN":
        flags.append(f"{check_name}_WARN: {check.detail}")

# Overall result — FAIL if any hard FAIL
overall = "FAIL" if any(f.endswith("FAIL") or "_FAIL:" in f for f in flags) else "PASS"

log.info(f"IDP result for {cheque_id}: {overall}, flags: {flags}")

return IDPResult(
    cheque_id           = cheque_id,
    amount_match        = amount_check,
    date_check          = date_check,
    ifsc_check          = ifsc_check,
    micr_account_match  = micr_check,
    overall_validation  = overall,
    flags               = flags
)
```

# ── FastAPI Endpoint ───────────────────────────────────────────────────────────

class IDPRequest(BaseModel):
cheque_id: str
fields: OCRFields

@router.post(”/validate”, response_model=IDPResult)
async def validate_idp(req: IDPRequest):
“”“Run IDP validation on extracted OCR fields”””
try:
result = run_idp_validation(req.cheque_id, req.fields)
return result
except Exception as e:
log.error(f”IDP validation failed for {req.cheque_id}: {e}”)
raise HTTPException(500, str(e))
