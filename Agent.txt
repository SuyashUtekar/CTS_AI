# ── MCP Tool Executor ─────────────────────────────────────────────────────────

def execute_mcp_tool(tool_name: str, tool_input: dict) -> str:
“””
Execute MCP tool call and return result as string.
In production: each tool calls its actual backend service.
“””
log.info(f”MCP tool called: {tool_name} with input: {tool_input}”)

```
if tool_name == "get_account_history":
    # Production: query PostgreSQL / core banking system
    account = tool_input.get("account_number", "")
    return json.dumps({
        "account_number": account,
        "avg_monthly_transaction": 45000,
        "max_single_transaction": 200000,
        "total_cheques_last_6m": 12,
        "suspicious_activity": False,
        "account_status": "ACTIVE"
    })

elif tool_name == "validate_ifsc":
    # Production: query RBI IFSC master database
    ifsc = tool_input.get("ifsc_code", "")
    return json.dumps({
        "ifsc": ifsc,
        "valid": True,
        "bank": "State Bank of India",
        "branch": "Andheri West",
        "city": "Mumbai",
        "active": True
    })

elif tool_name == "write_audit_log":
    # Production: write to Elasticsearch / WORM storage
    log.info(f"AUDIT LOG: {json.dumps(tool_input)}")
    return json.dumps({"logged": True, "log_id": f"LOG_{datetime.now().timestamp()}"})

elif tool_name == "publish_decision":
    # Production: publish to Kafka topic cheque.decided
    log.info(f"EVENT PUBLISHED: {json.dumps(tool_input)}")
    return json.dumps({"published": True, "topic": "cheque.decided"})

return json.dumps({"error": f"Unknown tool: {tool_name}"})
```

# ── Claude Agent with Tool Loop ───────────────────────────────────────────────

def run_claude_agent(cheque_id: str, ocr_fields: OCRFields,
idp_result: IDPResult, risk_result: RiskResult) -> AgentDecision:
“””
Run Claude Sonnet as decision agent with MCP tools.
Implements agentic tool-use loop.
“””
log.info(f”[Stage 6] Running Claude agent for: {cheque_id}”)

```
client  = anthropic.Anthropic(api_key=settings.anthropic_api_key)
prompt  = build_agent_prompt(cheque_id, ocr_fields, idp_result, risk_result)
messages = [{"role": "user", "content": prompt}]

final_decision = Decision.MANUAL_REVIEW
reasoning      = ""
confidence     = 0.80

# Agentic loop — Claude calls tools until it reaches a decision
max_iterations = 5
for iteration in range(max_iterations):
    log.debug(f"Agent iteration {iteration + 1}")

    response = client.messages.create(
        model      = "claude-sonnet-4-5",
        max_tokens = 1024,
        tools      = MCP_TOOLS,
        messages   = messages
    )

    # Check stop reason
    if response.stop_reason == "end_turn":
        # Extract final text response
        for block in response.content:
            if hasattr(block, "text"):
                reasoning = block.text
        break

    elif response.stop_reason == "tool_use":
        # Process all tool calls
        tool_results = []

        for block in response.content:
            if block.type == "tool_use":
                tool_result = execute_mcp_tool(block.name, block.input)
                tool_results.append({
                    "type":        "tool_result",
                    "tool_use_id": block.id,
                    "content":     tool_result
                })

        # Add assistant response + tool results to history
        messages.append({"role": "assistant", "content": response.content})
        messages.append({"role": "user",      "content": tool_results})

    else:
        break

# Parse decision from Claude's reasoning
reasoning_lower = reasoning.lower()
if "auto_approve" in reasoning_lower or "auto approve" in reasoning_lower:
    final_decision = Decision.AUTO_APPROVE
    confidence     = 0.95
    route          = "NGCB_EXPORT"
elif "reject" in reasoning_lower:
    final_decision = Decision.REJECT
    confidence     = 0.90
    route          = "REJECTED_QUEUE"
else:
    final_decision = Decision.MANUAL_REVIEW
    confidence     = 0.75
    route          = "MANUAL_REVIEW"

# Override: critical IDP failures always go to manual review
if idp_result.overall_validation == "FAIL" and final_decision == Decision.AUTO_APPROVE:
    final_decision = Decision.MANUAL_REVIEW
    route          = "MANUAL_REVIEW"
    reasoning     += "\n[Override: IDP validation failed — escalated to manual review]"

log.info(f"Agent decision for {cheque_id}: {final_decision} (confidence: {confidence})")

return AgentDecision(
    cheque_id  = cheque_id,
    decision   = final_decision,
    confidence = confidence,
    reasoning  = reasoning or f"Risk score {risk_result.risk_score} — {final_decision}",
    routed_to  = route,
    timestamp  = datetime.now()
)
```

# ── FastAPI Endpoint ───────────────────────────────────────────────────────────

class AgentRequest(BaseModel):
cheque_id: str
ocr_fields: OCRFields
idp_result: IDPResult
risk_result: RiskResult

@router.post(”/decide”, response_model=AgentDecision)
async def agent_decide(req: AgentRequest):
“”“Run Claude agent to make cheque clearing decision”””
if not settings.anthropic_api_key:
raise HTTPException(500, “ANTHROPIC_API_KEY not configured”)
try:
result = run_claude_agent(
req.cheque_id, req.ocr_fields,
req.idp_result, req.risk_result
)
return result
except Exception as e:
log.error(f”Agent decision failed for {req.cheque_id}: {e}”)
raise HTTPException(500, str(e))
